import json
import re

def cirilica_u_latinicu(tekst):
    mapa = {
        'А': 'A', 'Б': 'B', 'В': 'V', 'Г': 'G', 'Д': 'D', 'Ђ': 'Đ', 'Е': 'E', 'Ж': 'Ž', 'З': 'Z', 'И': 'I',
        'Ј': 'J', 'К': 'K', 'Л': 'L', 'Љ': 'Lj', 'М': 'M', 'Н': 'N', 'Њ': 'Nj', 'О': 'O', 'П': 'P', 'Р': 'R',
        'С': 'S', 'Т': 'T', 'Ћ': 'Ć', 'У': 'U', 'Ф': 'F', 'Х': 'H', 'Ц': 'C', 'Ч': 'Č', 'Џ': 'Dž', 'Ш': 'Š',
        'а': 'a', 'б': 'b', 'в': 'v', 'г': 'g', 'д': 'd', 'ђ': 'đ', 'е': 'e', 'ж': 'ž', 'з': 'z', 'и': 'i',
        'ј': 'j', 'к': 'k', 'л': 'l', 'љ': 'lj', 'м': 'm', 'н': 'n', 'њ': 'nj', 'о': 'o', 'п': 'p', 'р': 'r',
        'с': 's', 'т': 't', 'ћ': 'ć', 'у': 'u', 'ф': 'f', 'х': 'h', 'ц': 'c', 'ч': 'č', 'џ': 'dž', 'ш': 'š',
        'Ѝ': 'Ì', 'ѝ': 'ì', 'Ѐ': 'È', 'ѐ': 'è'
    }
    return "".join(mapa.get(c, c) for c in tekst)

def ocisti_rijec(tekst):
    očišćeno = re.sub(r'\s+[mfn]\b.*$', '', tekst.strip())
    return očišćeno

def ima_veliko_slovo(rijec):
    # Provjerava da li u riječi postoji bilo koje latinično veliko slovo (uključujući i naša sa kvačicama)
    velika_slova = "ABCDEFGHIJKLMNOPQRSTUVWXYZŠĐČĆŽÌÈ"
    return any(char in velika_slova for char in rijec)

def zadovoljava_uslov_za_e(rijec):
    r_mala = rijec.lower()
    
    if len(r_mala) < 3:
        return False
        
    varijacije_e = {'e', 'è', 'é', 'ê', 'ë', 'ē', 'ѐ'}
    
    if r_mala[0] in varijacije_e:
        return False
        
    if r_mala[-1] in varijacije_e:
        return False
        
    sredina = r_mala[1:-1]
    return any(v_e in sredina for v_e in varijacije_e)

def izvuci_ijekavske_akcente(ulazni_fajl, izlazni_txt):
    """
    IZVOR: Podaci se izvlače iz Kaikki.org rječnika ('kaikki.org-dictionary-SerboCroatian.jsonl').
    FILTRIRANJE: 
      1. Skupljaju se samo akcentovane riječi koje sadrže slovo 'e' u sredini (ne na početku/kraju).
      2. POTPUNO SE ODBACUJU sve riječi koje počinju ili sadrže veliko slovo (imena, skraćenice, naslovi).
    ZAPIS: Ćirilica se konvertuje u latinicu, brišu se gramatičke oznake i upisuje čista lista malih slova.
    """
    with open(ulazni_fajl, 'r', encoding='utf-8') as f_in, \
         open(izlazni_txt, 'w', encoding='utf-8') as f_out:
        
        upisane_rijeci = set()
        karakteri_akcenta = ['\u0300', '\u0301', '\u0304', '\u0309', '\u0311', '̏', '̀', '̑', '́', '̄', 'Ѝ', 'ѝ', 'Ѐ', 'ѐ']

        for linija in f_in:
            if not linija.strip():
                continue
                
            podaci = json.loads(linija)
            
            glavna_rijec = podaci.get('word', '')
            if any(akc in glavna_rijec for akc in karakteri_akcenta):
                cista_rijec = ocisti_rijec(cirilica_u_latinicu(glavna_rijec))
                
                # Dodat uslov: riječ ne smije da sadrži veliko slovo
                if cista_rijec and not ima_veliko_slovo(cista_rijec) and zadovoljava_uslov_za_e(cista_rijec):
                    if cista_rijec not in upisane_rijeci:
                        upisane_rijeci.add(cista_rijec)
                        f_out.write(f"{cista_rijec}\n")
            
            if 'forms' in podaci:
                for forma in podaci['forms']:
                    if 'form' in forma:
                        forma_tekst = forma['form']
                        
                        if any(akc in forma_tekst for akc in karakteri_akcenta):
                            cista_rijec = ocisti_rijec(cirilica_u_latinicu(forma_tekst))
                            
                            # Dodat uslov: riječ ne smije da sadrži veliko slovo
                            if cista_rijec and not ima_veliko_slovo(cista_rijec) and zadovoljava_uslov_za_e(cista_rijec):
                                if cista_rijec not in upisane_rijeci:
                                    upisane_rijeci.add(cista_rijec)
                                    f_out.write(f"{cista_rijec}\n")
                                
    print(f"Završeno! Uspješno je izvučeno ukupno {len(upisane_rijeci)} riječi (isključivo mala slova sa 'e' u sredini).")

izvuci_ijekavske_akcente('kaikki.org-dictionary-SerboCroatian.jsonl', 'akcentovane_rijeci.txt')
